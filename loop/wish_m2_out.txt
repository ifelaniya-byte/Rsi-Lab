def normalize_prompt(prompt: str) -> str:
    """Normalize prompt for deduplication: strip whitespace, lower case."""
    return prompt.strip().lower()

def deduplicate_tasks(tasks: List[Dict]) -> List[Dict]:
    """Drop tasks with duplicate normalized prompt hash."""
    seen = set()
    unique = []
    for task in tasks:
        norm = normalize_prompt(task["prompt"])
        h = hashlib.sha256(norm.encode()).hexdigest()
        if h not in seen:
            seen.add(h)
            unique.append(task)
    return unique

def gen_math_families(seed: int) -> Dict[str, Callable[[int], Tuple[str, str, str, str]]]:

    families = {}

    def integer_arithmetic(i: int) -> Tuple[str, str, str, str]:
        a = rng.randint(1, 100)
        b = rng.randint(1, 100)
        c = rng.randint(1, 100)
        op = rng.choice(['+', '-', '*', '/'])
        if op == '+':
            expr = f"{a} + {b} * {c}"
            answer = a + b * c
        elif op == '-':
            expr = f"{a} - {b} * {c}"
            answer = a - b * c
        elif op == '*':
            expr = f"{a} * {b} + {c}"
            answer = a * b + c
        else:  # '/'
            expr = f"{a} + {b} // {c}"
            answer = a + b // c
        prompt = f"Compute the value of {expr}. Return only the integer result."
        starter = prompt
        reference = str(answer)
        tests = ""  # math tasks have no tests in solutions.json
        return prompt, starter, reference, tests
    families["math_integer_arithmetic"] = integer_arithmetic

    # fractions family
    def fractions(i: int) -> Tuple[str, str, str, str]:
        denoms = [rng.randint(2, 10) for _ in range(3)]

def sample(solver_path: str, tasks: List[Dict], k: int = 3, timeout_per_task: int = 10) -> Dict[str, List[bool]]:
    """
    Grade the solver k times, returning per‑task pass matrix.
    Each sample is an independent model call (--no-cache).
    """
    # Filter to dev tasks (as per original grade.py behavior)
    dev_tasks = [t for t in tasks if t.get("dev")]
    if not dev_tasks:
        raise ValueError("No dev tasks in provided list")
    # Write temporary tasks.json for this sample
    import tempfile
    with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as f:
        json.dump(dev_tasks, f)
        temp_path = f.name
    try:
        matrix = {}
        for sample_idx in range(k):
            # Run grade.py with --no-cache
            cmd = [sys.executable, os.path.join(HARNESS_DIR, "grade.py"),
                   "--solver", solver_path,
                   "--tasks", temp_path,
                   "--timeout", str(timeout_per_task),
                   "--no-cache"]
            result = subprocess.run(cmd, capture_output=True, text=True, cwd=HARNESS_DIR)
            if result.returncode != 0:
                raise RuntimeError(f"grade.py failed: {result.stderr}")
            data = json.loads(result.stdout)
            for task in data["results"]:
                task_id = task["id"]
                if task_id not in matrix:
                    matrix[task_id] = []
                matrix[task_id].append(task["passed"])
        return matrix
    finally:
        os.unlink(temp_path)

def paired_bootstrap_lcb(deltas: List[float], seed: int = 42, replicates: int = 2000) -> float:
    """
    Compute 95% lower confidence bound of mean delta via paired bootstrap.
    Returns LCB (5th percentile of bootstrap distribution).
    """
    rng = random.Random(seed)
    n = len(deltas)
    means = []
    for _ in range(replicates):
        resampled = [deltas[rng.randint(0, n-1)] for _ in range(n)]
        means.append(statistics.mean(resampled))
    means.sort()
    # 5th percentile (lower bound of 90% CI -> 95% LCB)
    idx = int(0.05 * replicates)
    return means[idx]

def keep_decision(base_matrix: Dict[str, List[bool]],
                  cand_matrix: Dict[str, List[bool]],
                  seed: int = 42) -> Tuple[bool, Dict]:
    """
    Statistically honest keep gate.
    Returns (keep, report).
    """
    # Ensure same tasks and sample counts
    tasks = sorted(base_matrix.keys())
    if set(tasks) != set(cand_matrix.keys()):
        raise ValueError("Task sets differ")
    k_base = len(next(iter(base_matrix.values())))
    k_cand = len(next(iter(cand_matrix.values())))
    # Compute per‑task pass rates
    base_rates = {t: sum(base_matrix[t]) / k_base for t in tasks}
    cand_rates = {t: sum(cand_matrix[t]) / k_cand for t in tasks}
    deltas = [cand_rates[t] - base_rates[t] for t in tasks]
    # Overall LCB
    overall_lcb = paired_bootstrap_lcb(deltas, seed)
    # Per‑suite no‑regression
    suites = {}
    for t in tasks:
        # Infer suite from task id prefix
        if t.startswith("repair_"):
            suite = "code-repair"
        elif t.startswith("gen_"):
            suite = "code-gen"
        else:
            suite = "math"
        suites.setdefault(suite, []).append(deltas[tasks.index(t)])
    suite_lcbs = {}
    for suite, suite_deltas in suites.items():
        # Convert delta rates to task units: sum(delta) * len(suite_deltas) ? Wait.
        # Actually we need LCB of sum of deltas (in task units).
        # Each delta is rate difference; sum of deltas = total tasks gained.
        # Bootstrap the sum directly.
        sum_deltas = [sum(suite_deltas)]  # not correct for bootstrap
        # Better: bootstrap the mean delta, then multiply by n.
        n = len(suite_deltas)
        mean_lcb = paired_bootstrap_lcb(suite_deltas, seed + hash(suite))
        sum_lcb = mean_lcb * n
        suite_lcbs[suite] = sum_lcb
    # Decision
    keep = overall_lcb > 0 and all(lcb >= -1.0 for lcb in suite_lcbs.values())
    report = {
        "overall_lcb": overall_lcb,
        "suite_lcbs": suite_lcbs,
        "per_task_deltas": {t: deltas[i] for i, t in enumerate(tasks)},
        "n_tasks": len(tasks),
        "k_base": k_base,
        "k_cand": k_cand,
    }
    return keep, report

    # For simplicity, we'll skip execution in this test and assume correctness.
    # Instead, we'll check that each task has a reference and tests.
    all_pass = True
    detail = ""
    for task in tasks:
        tid = task["id"]
        if tid not in solutions:
