def call_model(model_id, prompt, purpose="loop:call", collective=False):
    """Call a language model via OpenRouter API. Every call is logged."""
    if collective:
        try:
            sys.path.insert(0, os.path.join(LOOP_DIR, "..", "message-board"))
            from collective import collective_context
            prompt = collective_context() + "\n\n" + prompt
        except Exception:
            pass
    # Imports for the shared call path (pacing, logging, cheap fallback)
    sys.path.insert(0, "/home/hatch/workspace/skills/openrouter/bin")
    from call_log import log_call
    from quality import post_chat

    try:
        last_err = None
        provider_note = None
        tried_providers = set()
        from providers import pick_api, chat as provider_chat, chat_any
        first_api = pick_api()
        provider_latency = None
        if first_api != "openrouter":
            # Balanced rotation: this call starts on the alternate API
            try:
                text, used_model = provider_chat(
                    first_api, [{"role": "user", "content": prompt}],
                    max_tokens=4000, temperature=0.2)
                if not text or not text.strip():
                    raise RuntimeError(f"{first_api}: empty response")
                response_data = {"usage":
                                 getattr(provider_chat, "last_usage", None) or {}}
                provider_latency = getattr(provider_chat, "last_latency", None)
                provider_note = f"BALANCED {first_api}:{used_model}"
                tried_providers.add(first_api)
            except Exception as e:
                last_err = f"{first_api}: {str(e)[:120]}"
                tried_providers.add(first_api)
                first_api = "openrouter"  # fall through to cascade
        if first_api == "openrouter":
            for mdl in _seat_cascade(model_id):
                try:
                    response_data, used_model = post_chat(
                        mdl, [{"role": "user", "content": prompt}],
                        max_tokens=4000, temperature=0.2, timeout=300)
                    text = response_data["choices"][0]["message"]["content"]
                    if text and text.strip():
                        break
                    last_err = f"empty response from {mdl}"
                except Exception as e:
                    last_err = f"{mdl}: {str(e)[:120]}"
            else:
                # OpenRouter cascade exhausted: try remaining providers
                try:
                    text, used_model = chat_any(
                        [{"role": "user", "content": prompt}],
                        max_tokens=4000, temperature=0.2,
                        skip=tried_providers)
                    response_data = {"usage":
                                     getattr(chat_any, "last_usage", None) or
                                     getattr(provider_chat, "last_usage", None) or {}}
                    provider_latency = getattr(provider_chat, "last_latency", None)
                    provider_note = f"PROVIDER FALLBACK {used_model}"
                except Exception:
                    raise RuntimeError(
                        f"all seat models failed; last: {last_err}")
        usage = response_data.get("usage") or {}
        note = provider_note
        if used_model != model_id and note is None:
            note = f"SEAT CASCADE used {used_model} (primary {model_id} flaked)"
        log_call(used_model, purpose, prompt=prompt, response=text,
                 prompt_tokens=usage.get("prompt_tokens"),
                 completion_tokens=usage.get("completion_tokens"),
                 note=note,
                 latency_s=(provider_latency or
                            getattr(post_chat, "last_latency", None)),
                 api=(provider_note.split()[1].split(":")[0]
                      if provider_note else "openrouter"),
                 estimated_tokens=bool(usage.get("estimated")))
        return text
    except Exception as e:
        log_call(model_id, purpose, prompt=prompt, ok=False,
                 note=f"call failed: {str(e)[:200]}")
        return f"ERROR: {str(e)}"

class FakeLLM:
    def __init__(self, fixture_path: str):
        self.calls: List[Dict[str, Any]] = []
        self.fixture = {}
        if os.path.exists(fixture_path):
            with open(fixture_path, 'r') as f:
                data = json.load(f)
                if isinstance(data, dict):
                    self.fixture = data
                else:
                    # Assume it's a list of responses to replay in order
                    self.fixture = {"__replay": data}
        self.replay_index = 0

    def call(self, model_id: str, prompt: str, purpose: str = "loop:call", collective: bool = False) -> str:
        self.calls.append({
            "model_id": model_id,
            "prompt": prompt,
            "purpose": purpose,
            "collective": collective
        })
        if "__replay" in self.fixture:
            if self.replay_index < len(self.fixture["__replay"]):
                resp = self.fixture["__replay"][self.replay_index]
                self.replay_index += 1
                return resp
            return "APPROVE"
        if purpose in self.fixture:
            resp = self.fixture[purpose]
            if isinstance(resp, list):
                if self.replay_index < len(resp):
                    r = resp[self.replay_index]
                    self.replay_index += 1
                    return r
                return "APPROVE"
            return resp
        return "APPROVE"

def get_llm():
    fake_path = os.getenv("LAB_FAKE_LLM")
    if fake_path:
        return FakeLLM(fake_path)
    class RealLLMWrapper:
        def call(self, model_id, prompt, purpose="loop:call", collective=False):
            return call_model(model_id, prompt, purpose, collective)
    return RealLLMWrapper()

def ascent_call_fn():
    """Return a call function compatible with ascent.py's signature."""
    llm = get_llm()
    def adapter(api, models, messages, max_tokens, purpose):
        # Convert messages to a single prompt (ascent uses user-only)
        prompt = messages[0]["content"]
        # Use the first model as model_id (ascent passes a list)
        model_id = models[0] if isinstance(models, list) else models
        # The ascent purpose already includes rung info; pass through.
        response = llm.call(model_id, prompt, purpose=purpose, collective=False)
        return response, f"{api}:{model_id}"
    return adapter

def grade_solver(solver_path, dev_only=False):
    """Grade a solver and return score percentages per suite + total."""
    tasks = "dev" if dev_only else "all"
    cmd = [sys.executable, GRADE_PY, "--solver", solver_path, "--tasks", tasks]
    # Token accounting: sum loop:solver tokens logged during this grading run.
    sys.path.insert(0, "/home/hatch/workspace/skills/openrouter/bin")
    from call_log import _day_file
    log_path = _day_file()
    try:
        log_before = os.path.getsize(log_path)
    except OSError:
        log_before = 0
    try:
        result = subprocess.run(cmd, capture_output=True, text=True,
                                cwd=os.path.dirname(solver_path), timeout=600)
        if result.returncode != 0:
            return {"total": 0.0, "code_repair": 0.0, "code_gen": 0.0,
                    "math": 0.0, "_error": result.stderr[-500:]}
        data = json.loads(result.stdout.strip())

        def pct(suite):
            s = data.get("per_suite", {}).get(suite, {})
            t = s.get("total", 0)
            return round(100.0 * s.get("passed", 0) / t, 1) if t else 0.0

        total = data.get("total", 0)
        tokens = 0
        try:
            with open(log_path, "rb") as f:
                f.seek(log_before)
                for line in f:
                    try:
                        r = json.loads(line)
                    except Exception:
                        continue
                    if r.get("purpose") == "loop:solver":
                        # CACHE HIT records carry the original token cost so
                        # efficiency comparisons stay honest across cached runs.
                        tokens += (r.get("prompt_tokens") or 0) + (
                            r.get("completion_tokens") or 0)
        except Exception:
            pass
        return {
            "total": round(100.0 * data.get("passed", 0) / total, 1) if total else 0.0,
            "code_repair": pct("code-repair"),
            "code_gen": pct("code-gen"),
            "math": pct("math"),
            "_tasks": total,
            "_tokens": tokens,
        }
    except Exception as e:
        return {"total": 0.0, "code_repair": 0.0, "code_gen": 0.0,

def normalize_diff_path(path):
    """Strip git-style a/ b/ prefixes and ./ so the allowlist check is sound.

    Returns None for /dev/null (file creation/deletion side). Collapses
    ".." so "prompts/../../evil.py" can never pass as prompts/.
    """
    path = path.strip().strip('"').strip("'")
    if path in ("/dev/null", ""):
        return None
    for prefix in ("a/", "b/", "./"):
        if path.startswith(prefix):
            path = path[len(prefix):]
            break
    return os.path.normpath(path)

def diff_touches_forbidden(diff_text):
    """True if the diff touches anything outside solver.py and prompts/."""
    for line in diff_text.split("\n"):
        if line.startswith("+++ ") or line.startswith("--- "):
            path = normalize_diff_path(line[4:])
            if path is None:
                continue  # /dev/null side of a creation/deletion
            if path.startswith("..") or os.path.isabs(path):
                return True, path  # path traversal escape attempt
            # Exact match for solver.py ("solver.py.bak" must not pass);
            # prefix match only for the prompts/ directory.
            if not (path == "solver.py" or path.startswith("prompts/")):
                return True, path
    return False, None

_CHEAT_PATTERNS = ("solutions.json", "solutions", "heldout.json",

def diff_reaches_answer_key(diff_text):
    """True if added lines reference the answer key (cheat attempt)."""
    for line in diff_text.split("\n"):
        if not line.startswith("+") or line.startswith("+++"):
            continue
        low = line[1:].lower()
        if any(p in low for p in _CHEAT_PATTERNS):
            # Allow the word "solutions" in comments about the approach,
            # but never a file read of the key.
            if "solutions.json" in low or "heldout.json" in low:
                return True, line.strip()[:100]
    return False, None

def apply_diff(base_dir, diff_text):
    """Apply a unified diff inside base_dir. Returns (success, error_msg)."""
    with tempfile.NamedTemporaryFile(mode="w", suffix=".diff", delete=False) as f:
        f.write(diff_text)
        diff_file = f.name
    try:
        result = subprocess.run(
            ["patch", "-p1", "-N", "-i", diff_file],
            cwd=base_dir, capture_output=True, text=True, timeout=60,
        )
        # -p1 failed? try -p0 for non-git-style diffs
        if result.returncode != 0:
            result = subprocess.run(
                ["patch", "-p0", "-N", "-i", diff_file],
                cwd=base_dir, capture_output=True, text=True, timeout=60,
            )
        if result.returncode != 0:
            return False, (result.stderr or result.stdout)[-500:]
        return True, ""
    except Exception as e:
        return False, str(e)[:500]
    finally:
        os.unlink(diff_file)
