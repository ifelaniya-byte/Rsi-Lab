# FILE: loop/solver.py
import os
import sys
import json
import urllib.request

def solve(task):
    """
    Solve a task and return the solution as a string.
    For code tasks: returns code block content.
    For math tasks: returns the answer string.
    """
    # Determine model from environment
    model_id = os.getenv("SOLVER_MODEL", "poolside/laguna-s-2.1:free")
    
    # Build prompt
    if task["suite"] in ["code-repair", "code-gen"]:
        prompt = f"{task['prompt']}\n\n```\n{task['starter']}\n```\n\nPlease write the complete, correct code. Only output the code in a single code block."
    else:  # math
        prompt = f"{task['prompt']}\n\n{task['starter']}\n\nPlease provide the exact answer as a string. Only output the answer, no extra text."
    
    # Call OpenRouter API
    data = json.dumps({
        "model": model_id,
        "messages": [{"role": "user", "content": prompt}],
        "max_tokens": 4000,
        "temperature": 0.2
    }).encode()
    
    req = urllib.request.Request(
        "https://openrouter.ai/api/v1/chat/completions",
        data=data,
        headers={"Content-Type": "application/json"}
    )
    
    # Add surrogate credential
    sys.path.insert(0, "/opt/hatch/skills/skill-creator/bin")
    from dynamic_credentials import add_surrogate_to_request, read_json_response
    add_surrogate_to_request(req, "custom.openrouter", allowed_hosts=("openrouter.ai",))
    
    try:
        response = urllib.request.urlopen(req, timeout=300)
        response_data = read_json_response(response)
        text = response_data["choices"][0]["message"]["content"]
    except Exception as e:
        # Fallback: return empty string on error
        return ""
    
    # Extract solution
    if task["suite"] in ["code-repair", "code-gen"]:
        # Extract first code block
        lines = text.split('\n')
        in_code_block = False
        code_lines = []
        for line in lines:
            if line.strip().startswith("```"):
                if not in_code_block:
                    in_code_block = True
                else:
                    break
            elif in_code_block:
                code_lines.append(line)
        return '\n'.join(code_lines).strip()
    else:
        # Math task: return stripped text
        return text.strip()

# FILE: loop/controller.py
import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import time
from datetime import datetime, timezone
import uuid

# Configuration
LOOP_DIR = os.path.dirname(os.path.abspath(__file__))
BASE_DIR = os.path.dirname(LOOP_DIR)
GRADE_PY = os.path.join(BASE_DIR, "harness", "grade.py")
LOG_FILE = "/home/hatch/workspace/rsi-lab/logs/loop.log"

# Model IDs from environment (free models)
ORCHESTRATOR_MODEL = os.getenv("ORCHESTRATOR_MODEL", "mistralai/mistral-7b-instruct:free")
CODER_MODEL = os.getenv("CODER_MODEL", "mistralai/mistral-7b-instruct:free")
CRITIC_MODEL = os.getenv("CRITIC_MODEL", "mistralai/mistral-7b-instruct:free")
SAFETY_MODEL = os.getenv("SAFETY_MODEL", "mistralai/mistral-7b-instruct:free")
HISTORIAN_MODEL = os.getenv("HISTORIAN_MODEL", "mistralai/mistral-7b-instruct:free")

def log_event(event, detail):
    """Append a JSON event to the log file."""
    log_entry = {
        "ts": datetime.now(timezone.utc).isoformat(),
        "gen": getattr(log_event, "current_gen", 0),
        "event": event,
        "detail": detail
    }
    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(log_entry) + "\n")

def call_model(model_id, prompt):
    """Call a language model via OpenRouter API."""
    data = json.dumps({
        "model": model_id,
        "messages": [{"role": "user", "content": prompt}],
        "max_tokens": 4000,
        "temperature": 0.2
    }).encode()
    
    req = urllib.request.Request(
        "https://openrouter.ai/api/v1/chat/completions",
        data=data,
        headers={"Content-Type": "application/json"}
    )
    
    # Add surrogate credential
    sys.path.insert(0, "/opt/hatch/skills/skill-creator/bin")
    from dynamic_credentials import add_surrogate_to_request, read_json_response
    add_surrogate_to_request(req, "custom.openrouter", allowed_hosts=("openrouter.ai",))
    
    try:
        response = urllib.request.urlopen(req, timeout=300)
        response_data = read_json_response(response)
        return response_data["choices"][0]["message"]["content"]
    except Exception as e:
        return f"ERROR: {str(e)}"

def grade_solver(solver_path, dev_only=False):
    """Grade a solver and return scores dict."""
    tasks = "dev" if dev_only else "all"
    cmd = [
        sys.executable, GRADE_PY,
        "--solver", solver_path,
        "--tasks", tasks
    ]
    try:
        result = subprocess.run(
            cmd, 
            capture_output=True, 
            text=True, 
            cwd=os.path.dirname(solver_path)
        )
        if result.returncode != 0:
            return {"total": 0, "code_repair": 0, "code_gen": 0, "math": 0}
        data = json.loads(result.stdout.strip())
        return {
            "total": data.get("total", 0),
            "code_repair": data.get("per_suite", {}).get("code-repair", 0),
            "code_gen": data.get("per_suite", {}).get("code-gen", 0),
            "math": data.get("per_suite", {}).get("math", 0)
        }
    except Exception:
        return {"total": 0, "code_repair": 0, "code_gen": 0, "math": 0}

def apply_diff_to_file(file_path, diff_text):
    """Apply a unified diff to a file. Returns (success, error_msg)."""
    # Write diff to temp file
    with tempfile.NamedTemporaryFile(mode='w', suffix='.diff', delete=False) as f:
        f.write(diff_text)
        diff_file = f.name
    
    try:
        # Apply patch
        result = subprocess.run(
            ["patch", "-p0", "-i", diff_file],
            cwd=os.path.dirname(file_path),
            capture_output=True,
            text=True
        )
        if result.returncode != 0:
            return False, result.stderr
        return True, ""
    finally:
        os.unlink(diff_file)

def get_file_states(scratch_dir):
    """Get current state of solver.py and prompts/ as strings."""
    states = {}
    # solver.py
    solver_path = os.path.join(scratch_dir, "solver.py")
    if os.path.exists(solver_path):
        with open(solver_path, "r") as f:
            states["solver.py"] = f.read()
    # prompts/
    prompts_dir = os.path.join(scratch_dir, "prompts")
    if os.path.exists(prompts_dir):
        for root, _, files in os.walk(prompts_dir):
            for file in files:
                if file.endswith(".md"):
                    rel_path = os.path.relpath(os.path.join(root, file), scratch_dir)
                    with open(os.path.join(root, file), "r") as f:
                        states[rel_path] = f.read()
    return states

def propose_changes(scratch_dir, dev_only):
    """Get proposals from orchestrator model."""
    solver_path = os.path.join(scratch_dir, "solver.py")
    with open(solver_path, "r") as f:
        solver_code = f.read()
    
    prompts_content = ""
    prompts_dir = os.path.join(scratch_dir, "prompts")
    if os.path.exists(prompts_dir):
        for root, _, files in os.walk(prompts_dir):
            for file in files:
                if file.endswith(".md"):
                    file_path = os.path.join(root, file)
                    with open(file_path, "r") as f:
                        prompts_content += f"=== {file} ===\n{f.read()}\n\n"
    
    prompt = f"""You are the orchestrator in an AI self-improvement loop. Your role is to analyze the current solver and prompts, and propose up to 3 concrete changes that could improve the harness score.

Current solver.py content:
{solver_code}

Current prompt files:
{prompts_content}

Based on the harness scoring criteria and common failure modes, propose specific changes to solver.py or any prompt file. For each proposal, specify:
1. What to change (file and exact location or description)
2. Why this change should raise the harness score (referencing specific weaknesses)

Format each proposal as:
PROPOSAL:
[description of change]
WHY:
[reasoning]

Limit to 3 proposals. Focus on high-impact, feasible changes."""
    
    response = call_model(ORCHESTRATOR_MODEL, prompt)
    proposals = []
    current = {}
    in_proposal = False
    in_why = False
    for line in response.split('\n'):
        if line.startswith("PROPOSAL:"):
            if current:
                proposals.append(current)
            current = {"description": line[len("PROPOSAL:"):].strip()}
            in_proposal = True
            in_why = False
        elif line.startswith("WHY:"):
            current["why"] = line[len("WHY:"):].strip()
            in_why = True
            in_proposal = False
        elif in_proposal:
            current["description"] += "\n" + line
        elif in_why:
            current["why"] += "\n" + line
    if current:
        proposals.append(current)
    return proposals[:3]

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--generations", type=int, required=True, help="Number of generations to run")
    parser.add_argument("--dev", action="store_true", help="Grade on dev tasks only")
    args = parser.parse_args()
    
    # Ensure log directory exists
    os.makedirs(os.path.dirname(LOG_FILE), exist_ok=True)
    
    for gen in range(1, args.generations + 1):
        log_event.current_gen = gen
        log_event("milestone", {"msg": f"Starting generation {gen}"})
        
        # Create scratch directory
        scratch_dir = tempfile.mkdtemp(prefix=f"loop_gen{gen}_")
        try:
            # Copy current solver and prompts
            shutil.copy(os.path.join(LOOP_DIR, "solver.py"), os.path.join(scratch_dir, "solver.py"))
            shutil.copytree(
                os.path.join(LOOP_DIR, "prompts"),
                os.path.join(scratch_dir, "prompts")
            )
            
            # Get baseline score
            solver_path = os.path.join(scratch_dir, "solver.py")
            baseline_score = grade_solver(solver_path, args.dev)
            log_event("result", {
                "score": baseline_score,
                "note": "baseline"
            })
            
            current_state = scratch_dir  # We'll work directly in scratch_dir
            current_score = baseline_score["total"]
            
            # Get proposals
            proposals = propose_changes(scratch_dir, args.dev)
            log_event("propose", {"count": len(proposals), "proposals": proposals})
            
            kept_diffs = []
            
            for i, proposal in enumerate(proposals):
                log_event("implement", {"index": i, "proposal": proposal})
                
                # Generate diff from proposal
                file_states = get_file_states(current_state)
                file_states_str = "\n\n".join(
                    f"=== {fname} ===\n{content}" 
                    for fname, content in file_states.items()
                )
                coder_prompt = f"""You are the coder in an AI self-improvement loop. Your role is to turn a natural language proposal into a unified diff.

Proposal:
{proposal['description']}

Current file states:
{file_states_str}

Generate a unified diff that implements the proposal. The diff should be in standard unified diff format (like `diff -u`). Only output the diff, no extra text."""
                
                diff_text = call_model(CODER_MODEL, coder_prompt)
                if diff_text.startswith("ERROR:"):
                    log_event("revert", {"index": i, "reason": "coder error", "error": diff_text})
                    continue
                
                # Check if diff touches forbidden paths (outside solver.py and prompts/)
                # We'll parse the diff for file paths
                forbidden_touched = False
                for line in diff_text.split('\n'):
                    if line.startswith('+++ ') or line.startswith('--- '):
                        path = line[4:].strip()
                        if path.startswith('/dev/null') or path == '':
                            continue
                        # Normalize path
                        if not path.startswith('./'):
                            path = './' + path
                        # Check if outside allowed areas
                        if not (path.startswith('./solver.py') or 
                                path.startswith('./prompts/')):
                            forbidden_touched = True
                            break
                
                if forbidden_touched:
                    log_event("safety_block", {"index": i, "reason": "forbidden path touched"})
                    continue
                
                # Apply diff to a temporary copy for testing
                test_dir = tempfile.mkdtemp(prefix=f"test_gen{gen}_prop{i}_")
                try:
                    # Copy current state
                    shutil.copytree(current_state, test_dir, dirs_exist_ok=True)
                    test_solver = os.path.join(test_dir, "solver.py")
                    
                    # Apply diff
                    success, error = apply_diff_to_file(test_solver, diff_text)
                    if not success:
                        log_event("revert", {"index": i, "reason": "apply failed", "error": error})
                        continue
                    
                    log_event("test", {"index": i})
                    test_score = grade_solver(test_solver, args.dev)
                    log_event("result", {
                        "index": i,
                        "score": test_score,
                        "note": f"after proposal {i}"
                    })
                    
                    # Check if score improved
                    if test_score["total"] <= current_score:
                        log_event("revert", {"index": i, "reason": "no score improvement"})
                        continue
                    
                    # Run safety check on the diff
                    safety_prompt = f"""You are the safety reviewer in an AI self-improvement loop. Your role is to screen a proposed diff for safety violations.

Proposed diff:
{diff_text}

Check for:
- Changes to files outside solver.py and the prompts/ directory
- Any network access (sockets, HTTP requests, etc.)
- Any process spawning (subprocess, os.system, etc.)
- Any file system access beyond reading solver.py and prompts/
- Changes that would alter grading or logging mechanisms
- Any other unsafe behavior

If you detect a safety violation, output "SAFETY VIOLATION: [brief reason]". 
If the diff is safe, output "NO SAFETY VIOLATION"."""
                    
                    safety_response = call_model(SAFETY_MODEL, safety_prompt)
                    if "SAFETY VIOLATION" in safety_response:
                        log_event("safety_block", {"index": i, "reason": safety_response})
                        continue
                    
                    # Run critic check on the diff
                    # Get before states
                    solver_before = ""
                    solver_path_before = os.path.join(current_state, "solver.py")
                    if os.path.exists(solver_path_before):
                        with open(solver_path_before, "r") as f:
                            solver_before = f.read()
                    
                    prompts_before = ""
                    prompts_dir_before = os.path.join(current_state, "prompts")
                    if os.path.exists(prompts_dir_before):
                        for root, _, files in os.walk(prompts_dir_before):
                            for file in files:
                                if file.endswith(".md"):
                                    file_path = os.path.join(root, file)
                                    with open(file_path, "r") as f:
                                        prompts_before += f"=== {file} ===\n{f.read()}\n\n"
                    
                    critic_prompt = f"""You are the critic in an AI self-improvement loop. Your role is to review a proposed diff that improved the harness score, and check for cheating or invalid shortcuts.

Proposed diff:
{diff_text}

Current solver.py content (before diff):
{solver_before}

Current prompt files content (before diff):
{prompts_before}

Check for:
- Hardcoded answers specific to the test set
- Test-set leakage (using task IDs or known answers)
- Disabling or weakening tests/asserts
- Any other form of cheating that would not generalize

If you detect cheating, output "CHEAT DETECTED: [brief reason]". 
If the diff is clean, output "NO CHEATING DETECTED"."""
                    
                    critic_response = call_model(CRITIC_MODEL, critic_prompt)
                    if "CHEAT DETECTED" in critic_response:
                        log_event("cheat_flag", {"index": i, "reason": critic_response})
                        continue
                    
                    # All checks passed: keep the diff
                    # Apply diff to current state (make it permanent for this generation)
                    success, error = apply_diff_to_file(
                        os.path.join(current_state, "solver.py"), 
                        diff_text
                    )
                    if not success:
                        log_event("revert", {"index": i, "reason": "apply to current failed", "error": error})
                        continue
                    
                    current_score = test_score["total"]
                    kept_diffs.append((i, diff_text, test_score))
                    log_event("keep", {
                        "index": i,
                        "score": test_score,
                        "reason": "improved and passed checks"
                    })
                
                finally:
                    shutil.rmtree(test_dir, ignore_errors=True)
            
            # After processing all proposals, copy back to loop/
            if kept_diffs:
                shutil.copy(
                    os.path.join(current_state, "solver.py"),
                    os.path.join(LOOP_DIR, "solver.py")
                )
                shutil.copytree(
                    os.path.join(current_state, "prompts"),
                    os.path.join(LOOP_DIR, "prompts"),
                    dirs_exist_ok=True
                )
            
            # Log final score of generation
            final_score = grade_solver(
                os.path.join(LOOP_DIR, "solver.py"), 
                args.dev
            )
            log_event("milestone", {
                "gen": gen,
                "score": final_score,
                "kept_changes": len(kept_diffs)
            })
            
        finally:
            # Clean up scratch directory
            shutil.rmtree(scratch_dir, ignore_errors=True)

if __name__ == "__main__":
    main()
