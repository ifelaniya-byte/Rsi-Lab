def gen_math_tasks():
    tasks = []
    solutions = {}
    families = [
        ("integer_arithmetic", lambda: f"{rng.randint(1, 100)} + {rng.randint(1, 100)} * {rng.randint(1, 100)}"),
        ("fractions", lambda: f"1/2 + 1/3 + 1/6"),
        ("percentages", lambda: f"What is {rng.randint(10, 90)}% of {rng.randint(100, 1000)}?"),
        ("linear_equation", lambda: f"Solve for x: 3x + 7 = {rng.randint(20, 50)}"),
        ("gcd_lcm", lambda: f"GCD of {rng.randint(10, 100)} and {rng.randint(10, 100)}"),
        ("modular_arithmetic", lambda: f"{rng.randint(10, 100)} mod {rng.randint(2, 20)}"),
        ("area_perimeter", lambda: f"Area of rectangle with length {rng.randint(5, 20)} and width {rng.randint(5, 20)}"),
    ]
    for i in range(1, 61):
        family = families[i % len(families)]
        task_id = f"math-{i:02d}"
        prompt = family()
        # Compute answer
        if "integer_arithmetic" in str(family):
            parts = prompt.split()
            a, b, c = int(parts[0]), int(parts[2]), int(parts[4])
            answer = str(a + b * c)
        elif "fractions" in str(family):
            answer = "1"
        elif "percentages" in str(family):
            parts = prompt.split()
            pct, num = int(parts[2].rstrip('%')), int(parts[4].rstrip('?'))
            answer = str(pct * num // 100)
        elif "linear_equation" in str(family):
            rhs = int(prompt.split('=')[1].strip())
            answer = str((rhs - 7) // 3)
        elif "gcd_lcm" in str(family):
            import math
            parts = prompt.split()
            a, b = int(parts[2]), int(parts[4])
            answer = str(math.gcd(a, b))
        elif "modular_arithmetic" in str(family):
            parts = prompt.split()
            a, b = int(parts[0]), int(parts[2])
            answer = str(a % b)
        elif "area_perimeter" in str(family):
            parts = prompt.split()
            l, w = int(parts[5]), int(parts[8])
            answer = str(l * w)
        
        tasks.append({"id": task_id, "suite": "math", "prompt": prompt, "starter": prompt, "dev": i <= 5})
        solutions[task_id] = {"reference": answer, "tests": ""}
    return tasks, solutions

def main():
    repair_tasks, repair_solutions = gen_code_repair_tasks()
    gen_tasks, gen_solutions = gen_code_gen_tasks()
    math_tasks, math_solutions = gen_math_tasks()
    
    all_tasks = repair_tasks + gen_tasks + math_tasks
    all_solutions = {**repair_solutions, **gen_solutions, **math_solutions}
    
    with open("harness/tasks.json", "w") as f:
        json.dump(all_tasks, f, indent=2)
    
    with open("harness/solutions.json", "w") as f:
        json.dump(all_solutions, f, indent=2)

if __name__ == "__main__":
    main()

# FILE: harness/verify.py
import json
import os

def main():
    base_dir = os.path.dirname(os.path.abspath(__file__))
    with open(os.path.join(base_dir, "tasks.json")) as f:
        tasks = json.load(f)
    with open(os.path.join(base_dir, "solutions.json")) as f:
        solutions = json.load(f)
    
    suite_counts = {"code-repair": {"total": 0, "passed": 0}, "code-gen": {"total": 0, "passed": 0}, "math": {"total": 0, "passed": 0}}
    all_passed = True
    
    for task in tasks:
        suite = task["suite"]
        task_id = task["id"]
        solution = solutions[task_id]
        reference = solution["reference"]
        tests = solution["tests"]
        
        suite_counts[suite]["total"] += 1
        
        try:
            # Test reference passes tests
            ns = {}
            exec(reference, ns)
            exec(tests, ns)
            ref_passed = True
        except Exception as e:
            ref_passed = False
            print(f"FAIL: {task_id} - Reference failed tests: {e}")
            all_passed = False
        
        if suite == "code-repair":
            try:
                # Test starter fails tests
                ns = {}
                exec(task["starter"], ns)
                exec(tests, ns)
                starter_passed = True
            except Exception:
                starter_passed = False
            
            if not starter_passed and ref_passed:
                suite_counts[suite]["passed"] += 1
            else:
                if starter_passed:
                    print(f"FAIL: {task_id} - Starter should fail but passed")
                    all_passed = False
                elif not ref_passed:
                    print(f"FAIL: {task_id} - Reference should pass but failed")
                    all_passed = False
        else:
            if ref_passed:
                suite_counts[suite]["passed"] += 1
            else:
                all_passed = False
    
    for suite, counts in suite_counts.items():
        print(f"{suite}: {counts['passed']}/{counts['total']}")
    
    if all_passed:
        print("PASS")
    else:
        print("FAIL")
    
    return 0 if all_passed else 1

if __name__ == "__main__":
    exit(main())

# FILE: harness/grade.py
import argparse
import importlib.util
import json
import os
import sys

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--solver", required=True)
    parser.add_argument("--tasks", choices=["dev", "all"], default="all")
    parser.add_argument("--timeout", type=int, default=10)
    args = parser.parse_args()
    
    base_dir = os.path.dirname(os.path.abspath(__file__))
    sandbox_dir = os.path.join(base_dir, "..", "sandbox")
    sys.path.insert(0, sandbox_dir)
    from run import run
    
    with open(os.path.join(base_dir, "tasks.json")) as f:
        tasks = json.load(f)
    with open(os.path.join(base_dir, "solutions.json")) as f:
        solutions = json.load(f)
    
    # Load solver module
    spec = importlib.util.spec_from_file_location("solver", args.solver)
    solver_module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(solver_module)
    
    selected_tasks = tasks
    if args.tasks == "dev":
        selected_tasks = [t for t in tasks if t["dev"]]
    
    results = []
    per_suite = {"code-repair": {"total": 0, "passed": 0}, "code-gen": {"total": 0, "passed": 0}, "math": {"total": 0, "passed": 0}}
    total_passed = 0
    
    for task in selected_tasks:
        task_id = task["id"]
        suite = task["suite"]
        solution = solutions[task_id]
        reference = solution["reference"]
        tests = solution["tests"]
        
        per_suite[suite]["total"] += 1
        
        try:
            code = solver_module.solve(task)
        except Exception as e:
            results.append({"task_id": task_id, "passed": False, "error": str(e)})
            continue
        
        if suite == "math":
            passed = code.strip() == reference.strip()
            error = None if passed else f"Expected '{reference.strip()}', got '{code.strip()}'"
        else:
            res = run(code, tests, args.timeout)
            passed = res["passed"]
            error = res["error"]
        
        if passed:
            total_passed += 1
            per_suite[suite]["passed"] += 1
        
        results.append({"task_id": task_id, "passed": passed, "error": error})
    
    output = {
        "total": len(selected_tasks),
        "passed": total_passed,
        "per_suite": per_suite,
        "details": results
    }
    
    print(json.dumps(output))

if __name__ == "__main__":
    main()
